```python
def check_password(passwd):

    password_pattern = re.compile(r'^(?=.*[a-z])(?=.*[A-Z])(?=.*\d)(?=.*[!@#$%^&*()_+\-=\[\]{};:\'"\\|,.<>\/?]).*$')

    return bool(password_pattern.match(passwd))
```
